'use strict';
angular.module('eehNavigation')
    .directive('eehNavigationActiveMenuItem', ActiveMenuItemDirective)
    .directive('eehNavigationAltParent', AltParentDirective);

function isMenuItemActive(menuItem, $state) {
    if (!menuItem.hasChildren()) {
        // Return whether or not the leaf is active.
        // return angular.isDefined(menuItem.state) && $state.includes(menuItem.state);
        return false;
    }
    var children = menuItem.children();
    for (var i = 0; i < children.length; i++) {
        if (angular.isDefined(children[i].state) && $state.includes(children[i].state)) {
            // Return true if the child menu item is active.
            return true;
        }
        if (isMenuItemActive(children[i], $state)) {
            // Return true if the child has at least one active child of its own.
            return true;
        }
    }
    // Return false if the menu item has no active children.
    return false;
}

function hasActiveChildren(menuItem, $state) {
    if (!menuItem.hasChildren()) {
        return false;
    }
    var children = menuItem.children();
    for (var i = 0; i < children.length; i++) {
        if(children[i].activeNod || (angular.isDefined(children[i].state) && $state.includes(children[i].state))) {
            return true;
        }
        if(hasActiveChildren(children[i], $state)) {
            return true;
        }
    }
    return false;
}

/** @ngInject */
function ActiveMenuItemDirective($state, $rootScope) {
    return {
        restrict: 'A',
        scope: {
            menuItem: '=eehNavigationActiveMenuItem'
        },
        link: function (scope, element) {
            var checkIsActive = function (isCollapsed) {
                var isActive = isMenuItemActive(scope.menuItem, $state);
                element.toggleClass('active', isActive);

                setTimeout(function(){
                    var isElementTheActiveChild = scope.menuItem && !scope.menuItem.hasChildren() && (scope.menuItem.activeNod || (angular.isDefined(scope.menuItem.state) && $state.includes(scope.menuItem.state))); 
                    element.toggleClass('alt-active-sidebar-menuitem', isElementTheActiveChild ? true : false);

                    if(scope.menuItem.activeNod || (angular.isDefined(scope.menuItem.state) && $state.includes(scope.menuItem.state))) {
                        console.log((angular.isDefined(scope.menuItem.state) && $state.includes(scope.menuItem.state)), scope.menuItem.activeNod);
                    }
                    
                    if(isElementTheActiveChild) {
                        $rootScope.$broadcast('altTheActiveChildIsSet', {'state': $state, 'isCollapsed': isCollapsed});

                        console.log("isActive", scope.menuItem);
                    }
                }, 10);
            };

            scope.$on("$stateChangeSuccess", function($event, isCollapsed){
                checkIsActive(isCollapsed);
            });
            scope.$on("menuStateChanged", function($event, isCollapsed){
                checkIsActive(isCollapsed);
            });
            scope.$on("menuSidebarChanged", function($event, isCollapsed){
                checkIsActive(isCollapsed);
            });
            checkIsActive();
        }
    };
}

/** @ngInject */
function AltParentDirective() {
    return {
        restrict: "A",
        scope: {
            menuItem: "=eehNavigationAltParent"
        },
        link: function(scope, element) {
            function checkActiveChild($event, params) {
                setTimeout(function(){
                    var hasActive = hasActiveChildren(scope.menuItem, params.state);
                    var setClass = "";
                    if(hasActive) {
                        setClass = params.isCollapsed ? "sidebar-active-collapsed" : "sidebar-active-parent";
                    }
                    scope.menuItem.className = setClass;

                    if(hasActive) {
                        console.log("is parent", scope.menuItem, params);
                    }

                    element.toggleClass("sidebar-active-parent", hasActive);
                    element.toggleClass("sidebar-active-collapsed", hasActive && params.isCollapsed);
                }, 10);
            }

            scope.$on("altTheActiveChildIsSet", function($event, params) {
                checkActiveChild($event, params);
            });
            
            // scope.$on("$stateChangeSuccess", function($event, isCollapsed){
            //     checkActiveChild(null, false);
            // });
            // scope.$on("menuStateChanged", function($event, isCollapsed){
            //     checkActiveChild($event, isCollapsed);
            // });
            // scope.$on("menuSidebarChanged", function($event, isCollapsed){
            //     checkActiveChild($event, isCollapsed);
            // });
            // checkActiveChild(null, false);
        }
    };
}
